# for문 이용
# [장점] : 메모리가 적고, 실행속도가 빠름
# [단점] : 코드가 복잡해질 수 있음
def fib_loop(n):
    if n in [1,2]:
        return 1
    a = b = 1
    for n in range(n-2):
        a, b = b, a+b
    return b

# 재귀함수 이용
# [장점] : 코드가 직관적이고, 이해하기 쉬움
# [단점] : 메모리 스택 오버플로우나 프로그램 속도가 느려질 수 있음
def fib(n):
    if n in [1,2]:
        return 1
    return fib(n-1) + fib(n-2)

<실행 시간 비교> : loop가 더 빠름
from timeit import default_timer as timer
from datetime import timedelta
start = timer()
fib_loop(30)
end = timer()
print(timedelta(seconds=end-start))

class Circle:
    def __init__(self, a, b, c):
        self.a = a
        self.b = b
        self.c = c

    def area(self):
        pi = 3.14
        print(pi*self.a**2)

    def circumference(self):
        pi = 3.14
        print(2*pi*self.a)

    def center(self):
        return self.b, self.c

c1 = Circle(3, 2, 4)
c1.area()
c1.circumference() 
c1.center()
