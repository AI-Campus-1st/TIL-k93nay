[과제 3번 문항 답]

bloods = [
    'A', 'B', 'A', 'O', 'AB', 'AB',
    'O', 'A', 'B', 'O', 'B', 'AB',
]

blood_types = {}
for type in bloods:
    if type in blood_types:
        blood_types[type] += 1
    else:
        blood_types[type] = 1
print(blood_types)
